// NppExec script for building Icosahedron project
// Save this as a script in NppExec (F6 -> Save)

// Change to project directory
cd "$(CURRENT_DIRECTORY)"

// If we're in src folder, go up one level
IF "$(CURRENT_DIRECTORY)" == "$(CURRENT_DIRECTORY:\src=)" THEN
    cd "$(CURRENT_DIRECTORY)\.."
ENDIF

// Check for g++ (MinGW)
cmd /c "where g++ >nul 2>&1"
IF $(EXITCODE) != 0 THEN
    // Try w64devkit if in raylib folder
    IF EXIST "C:\raylib\w64devkit\bin\g++.exe" THEN
        SET PATH=C:\raylib\w64devkit\bin;$(PATH)
    ENDIF
ENDIF

// Build using Makefile
IF EXIST "Makefile" THEN
    cmd /c "mingw32-make clean 2>nul"
    cmd /c "mingw32-make"
ELSE
    // Direct compilation if no Makefile
    cmd /c "g++ -std=c++17 -O2 -I C:/raylib/include src/main.cpp -L C:/raylib/lib -lraylib -lwinmm -lgdi32 -o bin/Icosahedron.exe"
    cmd /c "copy C:\raylib\lib\raylib.dll bin\ 2>nul"
ENDIF
